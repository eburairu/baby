frontend/components/ui/HexagonProgressArc.tsx を新規作成する。

このコンポーネントは Hexagon の clip-path の**外側**に重ねる SVG オーバーレイ。
六角形の外周パスを stroke-dasharray で部分表示することでプログレスストロークを実現する。

実装の要点:
1. Hexagon と同じ SVG viewBox・サイズ計算を使う（pointy=true 固定）
2. 六角形パスを pathLength="1" で定義 → stroke-dasharray で 0 から 1 まで制御
3. strokeWidth=5、fill="none"
4. stroke色は INDICATOR_COLORS[getIndicatorColor(progress)] を使用
5. isOverThreshold=true のとき className に "animate-pulse" を追加（Tailwind）
6. 絶対配置で Hexagon の上に重ねる（position: absolute, top: 0, left: 0）
7. progress を 1 にクランプして表示（Math.min(progress, 1)）

```typescript
'use client'

import { useMemo } from 'react'
import { INDICATOR_COLORS, getIndicatorColor } from '@/lib/indicatorUtils'

interface HexagonProgressArcProps {
  size: number
  progress: number
  isOverThreshold: boolean
  className?: string
}

export function HexagonProgressArc({
  size,
  progress,
  isOverThreshold,
  className,
}: HexagonProgressArcProps) {
  const { svgWidth, svgHeight, viewBox, pathD } = useMemo(() => {
    const cornerRadius = 12
    const w = size * Math.sqrt(3) / 2
    const h = size
    const cx = size / 2
    const cy = size / 2

    // pointy-top 六角形の6頂点（i=0 が上頂点 = 12時）
    const vertices = Array.from({ length: 6 }, (_, i) => {
      const angle = (Math.PI / 180) * (60 * i - 90)
      return {
        x: cx + (size / 2) * Math.cos(angle),
        y: cy + (size / 2) * Math.sin(angle),
      }
    })

    // 角丸六角形のSVGパス
    // 各辺: 頂点から cornerRadius 分手前で曲線開始 → 次の頂点へ
    const segments: string[] = []
    for (let i = 0; i < 6; i++) {
      const curr = vertices[i]
      const next = vertices[(i + 1) % 6]
      const prev = vertices[(i + 5) % 6]

      const fromPrev = {
        x: curr.x + (next.x - curr.x) * 0,
        y: curr.y + (next.y - curr.y) * 0,
      }
      // 前の頂点から curr への方向ベクトル
      const dxIn = curr.x - prev.x
      const dyIn = curr.y - prev.y
      const lenIn = Math.sqrt(dxIn * dxIn + dyIn * dyIn)
      const p1 = {
        x: curr.x - (dxIn / lenIn) * cornerRadius,
        y: curr.y - (dyIn / lenIn) * cornerRadius,
      }
      // curr から next への方向ベクトル
      const dxOut = next.x - curr.x
      const dyOut = next.y - curr.y
      const lenOut = Math.sqrt(dxOut * dxOut + dyOut * dyOut)
      const p2 = {
        x: curr.x + (dxOut / lenOut) * cornerRadius,
        y: curr.y + (dyOut / lenOut) * cornerRadius,
      }

      if (i === 0) {
        segments.push(`M ${p2.x} ${p2.y}`)
      } else {
        segments.push(`L ${p1.x} ${p1.y}`)
        segments.push(`Q ${curr.x} ${curr.y} ${p2.x} ${p2.y}`)
      }
      if (i === 5) {
        // 閉じる: 最後の辺の終端 (p1 of i=0) へ戻って Q
        const first = vertices[0]
        const nextFirst = vertices[1]
        const dxIn0 = first.x - vertices[5].x
        const dyIn0 = first.y - vertices[5].y
        const len0 = Math.sqrt(dxIn0 * dxIn0 + dyIn0 * dyIn0)
        const q1 = {
          x: first.x - (dxIn0 / len0) * cornerRadius,
          y: first.y - (dyIn0 / len0) * cornerRadius,
        }
        segments.push(`L ${q1.x} ${q1.y}`)
        // M で始まった p2 of i=0 へ Q で閉じる
        const dxOut0 = nextFirst.x - first.x
        const dyOut0 = nextFirst.y - first.y
        const len00 = Math.sqrt(dxOut0 * dxOut0 + dyOut0 * dyOut0)
        const startPoint = {
          x: first.x + (dxOut0 / len00) * cornerRadius,
          y: first.y + (dyOut0 / len00) * cornerRadius,
        }
        segments.push(`Q ${first.x} ${first.y} ${startPoint.x} ${startPoint.y}`)
      }
    }

    return {
      svgWidth: w,
      svgHeight: h,
      viewBox: `${(size - w) / 2} 0 ${w} ${h}`,
      pathD: segments.join(' '),
    }
  }, [size])

  const clampedProgress = Math.min(progress, 1)
  const color = INDICATOR_COLORS[getIndicatorColor(progress)]
  const pulseClass = isOverThreshold ? 'animate-pulse' : ''

  return (
    <svg
      width={svgWidth}
      height={svgHeight}
      viewBox={viewBox}
      style={{ position: 'absolute', top: 0, left: '50%', transform: 'translateX(-50%)', pointerEvents: 'none' }}
      className={`${pulseClass} ${className ?? ''}`.trim()}
    >
      <path
        d={pathD}
        fill="none"
        stroke={color}
        strokeWidth={5}
        pathLength={1}
        strokeDasharray={`${clampedProgress} 1`}
        strokeDashoffset={0}
        strokeLinecap="round"
      />
    </svg>
  )
}
```

注意: SVGパスの計算は hexagon.tsx と完全一致させる必要がある。hexagon.tsx の実際のパス計算コードを参照して、HexagonProgressArc でも同じ計算ロジックを使うこと（コードを読んで必要に応じて調整）。
上記コードは設計の骨格であり、実際の hexagon.tsx のパス計算と一致させるため実行時に調整が必要な場合がある。
